cmake_minimum_required(VERSION 3.24)
project(VideoTime VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Multimedia Test)
add_library(videotime_core src/domain.cpp src/overlay.cpp src/project.cpp src/media.cpp src/exportqueue.cpp)
target_include_directories(videotime_core PUBLIC src)
target_link_libraries(videotime_core PUBLIC Qt6::Core Qt6::Gui)
qt_add_executable(VideoTime WIN32 src/main.cpp src/mainwindow.cpp src/preview.cpp src/appearance.cpp)
target_link_libraries(VideoTime PRIVATE videotime_core Qt6::Widgets Qt6::Multimedia)
qt_add_resources(VideoTime ui_fonts PREFIX /fonts BASE assets/fonts FILES assets/fonts/NotoSansThai-Regular.ttf)
qt_add_resources(VideoTime ui_icons PREFIX /icons BASE assets/icons FILES assets/icons/chevron-down.svg assets/icons/chevron-up.svg assets/icons/check.svg)
if(MSVC)
    target_compile_options(videotime_core PRIVATE /W4 /utf-8)
    target_compile_options(VideoTime PRIVATE /W4 /utf-8)
else()
    target_compile_options(videotime_core PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(VideoTime PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    qt_add_executable(videotime_tests tests/core_test.cpp)
    target_link_libraries(videotime_tests PRIVATE videotime_core Qt6::Test Qt6::Widgets)
    qt_add_executable(videotime_ui_tests tests/ui_test.cpp src/mainwindow.cpp src/preview.cpp src/appearance.cpp)
    target_link_libraries(videotime_ui_tests PRIVATE videotime_core Qt6::Test Qt6::Widgets Qt6::Multimedia)
    qt_add_resources(videotime_ui_tests ui_test_fonts PREFIX /fonts BASE assets/fonts FILES assets/fonts/NotoSansThai-Regular.ttf)
    qt_add_resources(videotime_ui_tests ui_test_icons PREFIX /icons BASE assets/icons FILES assets/icons/chevron-down.svg assets/icons/chevron-up.svg assets/icons/check.svg)
    add_test(NAME ui COMMAND videotime_ui_tests)
    add_test(NAME core COMMAND videotime_tests)
    set_tests_properties(core ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_FONTDIR=C:/Windows/Fonts")
endif()
install(TARGETS VideoTime RUNTIME DESTINATION .)
install(FILES README.md THIRD-PARTY-NOTICES.md DESTINATION .)
install(DIRECTORY licenses docs DESTINATION .)
qt_generate_deploy_app_script(TARGET VideoTime OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
install(CODE "set(QT_DEPLOY_BIN_DIR .)")
install(SCRIPT ${deploy_script})
set(VIDEOTIME_FFMPEG_DIR "" CACHE PATH "Directory containing a redistributable FFmpeg build")
if(VIDEOTIME_FFMPEG_DIR)
    install(DIRECTORY "${VIDEOTIME_FFMPEG_DIR}/" DESTINATION tools FILES_MATCHING
        PATTERN "*.exe" PATTERN "*.dll" PATTERN "*.txt" PATTERN "*LICENSE*" PATTERN "*COPYING*")
endif()
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "VideoTime-${PROJECT_VERSION}-windows-x64")
include(CPack)
